name: Promote exact digest to M1

on:
  workflow_dispatch:
    inputs:
      digest:
        description: "Accepted OCI digest (sha256:...)"
        required: true
        type: string

permissions:
  contents: read
  packages: write
  actions: read

concurrency:
  group: promote-m1
  cancel-in-progress: false

jobs:
  promote:
    runs-on: ubuntu-24.04
    # Configure required reviewers for this environment in repository settings.
    environment: stable-promotion
    steps:
      - name: Require protected stable-promotion environment
        env:
          GH_TOKEN: ${{ github.token }}
        shell: bash
        run: |
          set -euo pipefail
          reviewers="$(gh api \
            -H 'Accept: application/vnd.github+json' \
            -H 'X-GitHub-Api-Version: 2026-03-10' \
            "repos/$GITHUB_REPOSITORY/environments/stable-promotion" \
            --jq '[.protection_rules[]? | select(.type == "required_reviewers") | .reviewers[]?] | length')"
          if [[ ! "$reviewers" =~ ^[0-9]+$ || "$reviewers" -lt 1 ]]; then
            echo 'stable-promotion must exist with at least one required reviewer; refusing promotion.' >&2
            exit 1
          fi

      - name: Validate requested identity
        env:
          DIGEST: ${{ inputs.digest }}
        run: |
          set -euo pipefail
          [[ "$DIGEST" =~ ^sha256:[0-9a-f]{64}$ ]]

      - name: Install transport and signature tooling
        run: |
          sudo apt-get update
          sudo apt-get install -y skopeo

      - uses: sigstore/cosign-installer@6f9f17788090df1f26f669e9d70d6ae9567deba6 # v4.1.2
        with:
          cosign-release: 'v3.0.6'

      - name: Login to GHCR
        env:
          GHCR_USER: ${{ github.actor }}
          GHCR_TOKEN: ${{ github.token }}
        run: |
          mkdir -p "$HOME/.docker"
          printf '%s' "$GHCR_TOKEN" | skopeo login --authfile "$HOME/.docker/config.json" \
            ghcr.io --username "$GHCR_USER" --password-stdin

      - name: Verify accepted immutable candidate
        env:
          DIGEST: ${{ inputs.digest }}
        run: |
          set -euo pipefail
          skopeo inspect --authfile "$HOME/.docker/config.json" "docker://ghcr.io/krism-eu/krisos45@${DIGEST}" >/dev/null
          cosign verify \
            --certificate-identity 'https://github.com/krism-eu/krisOS45/.github/workflows/build-m1.yml@refs/heads/main' \
            --certificate-oidc-issuer 'https://token.actions.githubusercontent.com' \
            "ghcr.io/krism-eu/krisos45@${DIGEST}"

      - name: Promote exact digest without rebuild
        env:
          DIGEST: ${{ inputs.digest }}
        run: |
          set -euo pipefail
          source="docker://ghcr.io/krism-eu/krisos45@${DIGEST}"
          target="docker://ghcr.io/krism-eu/krisos45:m1"
          skopeo copy --all --authfile "$HOME/.docker/config.json" "$source" "$target"
          promoted="$(skopeo inspect --authfile "$HOME/.docker/config.json" --format '{{.Digest}}' "$target")"
          test "$promoted" = "$DIGEST"
          printf 'Promoted exact accepted digest: %s\n' "$DIGEST"
